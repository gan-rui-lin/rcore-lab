## 一、问题域：上下文切换的架构差异

内核的上下文切换（context switch）将当前任务的 CPU 状态保存到内存，然后从内存恢复另一个任务的状态。核心差异在于：

1. **需要保存的寄存器集合不同**：各架构的 callee-saved 寄存器数量和名字不同
2. **线程指针机制不同**：RISC-V 用 tp 寄存器，x86 用 FS_BASE MSR，ARM 用 TPIDR_EL1
3. **页表切换指令不同**：satp vs CR3 vs TTBR0_EL1 vs CSR PGDL
4. **栈帧约定不同**：ABI 规范不同

**关键洞察**：上下文切换只需保存 **callee-saved 寄存器**（被调用者保存的寄存器），因为 caller-saved 寄存器由编译器在调用 `context_switch()` 前自动保存在栈上。

## 二、抽象层设计

### 2.1 KContext 结构体 + KContextArgs 枚举

```rust
// arch/src/lib.rs
#[cfg(feature = "kcontext")]
pub enum KContextArgs {
    KSP,    // 内核栈指针
    KTP,    // 内核线程指针
    KPC,    // 内核程序计数器（恢复后跳转地址）
}
```

每个架构定义自己的 `KContext` 结构体，并实现 `Index<KContextArgs>` 和 `IndexMut<KContextArgs>`。

### 2.2 统一接口

```rust
// 各架构导出的函数（签名相同，实现不同）
pub fn context_switch(current: &mut KContext, next: &KContext);
pub fn context_switch_pt(current: &mut KContext, next: &KContext, pt_token: usize);
pub fn read_current_tp() -> usize;
```

`context_switch` 只切换 CPU 寄存器，`context_switch_pt` 额外切换页表。

## 三、各架构 KContext 实现对比

### 3.1 RISC-V 64

```rust
// arch/src/riscv64/kcontext.rs
pub struct KContext {
    ksp: usize,              // 内核栈指针
    ktp: usize,              // 线程指针 (tp)
    _sregs: [usize; 12],     // s0-s11 (12 个 callee-saved)
    kpc: usize,              // 返回地址 (ra)
}
// 总大小: 15 * 8 = 120 bytes
```

**context_switch 汇编**：
```asm
# 保存当前上下文到 a0 (current)
sd   sp, 0*8(a0)    # ksp
sd   tp, 1*8(a0)    # ktp
sd   s0, 2*8(a0)    # _sregs[0]
sd   s1, 3*8(a0)
...
sd   s11, 13*8(a0)
sd   ra, 14*8(a0)   # kpc

# 恢复下一个上下文从 a1 (next)
ld   sp, 0*8(a1)
ld   tp, 1*8(a1)
ld   s0, 2*8(a1)
...
ld   s11, 13*8(a1)
ld   ra, 14*8(a1)
ret                   # 跳转到 next.kpc
```

**context_switch_pt**（额外切换 satp）：
```asm
# 在 context_switch 基础上增加：
srli a2, a2, 12           # 物理页号
li   t1, 8 << 60          # Sv39 mode
or   a2, a2, t1
csrw satp, a2             # 切换页表
sfence.vma                # 刷新 TLB
```

### 3.2 AArch64

```rust
// arch/src/aarch64/kcontext.rs
pub struct KContext {
    ksp: usize,              // 栈指针
    ktp: usize,              // TPIDR_EL1
    _sregs: [usize; 11],     // x19-x29 (11 个 callee-saved)
    kpc: usize,              // x30 (LR)
}
// 总大小: 14 * 8 = 112 bytes
```

**context_switch 汇编**：
```asm
# 使用 STP/LDP 成对存取（ARM 特有高效指令）
mov  x2, sp
mrs  x3, tpidr_el1
stp  x2, x3, [x0, 0*8]     # 保存 ksp, ktp
stp  x19, x20, [x0, 2*8]
stp  x21, x22, [x0, 4*8]
...
stp  x29, x30, [x0, 12*8]  # x30 = LR = kpc

ldp  x2, x3, [x1, 0*8]     # 恢复 ksp, ktp
mov  sp, x2
msr  tpidr_el1, x3
ldp  x19, x20, [x1, 2*8]
...
ldp  x29, x30, [x1, 12*8]
ret                          # 跳转到 next.kpc (x30)
```

**context_switch_pt**（额外切换 TTBR0）：
```asm
msr  ttbr0_el1, x2           # 切换页表
tlbi vmalle1                  # 刷新 TLB
dsb  sy                       # 数据同步屏障
isb                           # 指令同步屏障
```

### 3.3 x86-64

```rust
// arch/src/x86_64/kcontext.rs
pub struct KContext {
    ksp: usize,              // 栈指针
    ktp: usize,              // FS_BASE MSR
    rbx: usize,              // callee-saved
    rbp: usize,
    r12: usize,
    r13: usize,
    r14: usize,
    r15: usize,
    kpc: usize,              // 返回地址
}
// 总大小: 9 * 8 = 72 bytes
```

**context_switch_pt 汇编**：
```asm
# x86-64 使用 MSR 保存/恢复 TLS
pop  r8                       # 获取返回地址（naked 函数特有）

# 保存 FS_BASE (TLS)
mov  ecx, 0xC0000100          # IA32_FS_BASE MSR
rdmsr
shl  rdx, 32
or   rax, rdx
mov  [rdi + 1*8], rax         # 保存到 ktp

# 保存 callee-saved 寄存器
mov  [rdi + 0*8], rsp
mov  [rdi + 2*8], rbx
mov  [rdi + 3*8], rbp
mov  [rdi + 4*8], r12
mov  [rdi + 5*8], r13
mov  [rdi + 6*8], r14
mov  [rdi + 7*8], r15
mov  [rdi + 8*8], r8          # 保存返回地址

# 切换页表
mov  cr3, rdx                  # CR3 = page_table

# 恢复
mov  rsp, [rsi + 0*8]
mov  rbx, [rsi + 2*8]
...
mov  r15, [rsi + 7*8]

# 恢复 FS_BASE
mov  rax, [rsi + 1*8]
mov  ecx, 0xC0000100
wrmsr

push [rsi + 8*8]              # 推入返回地址
ret                            # 跳转
```

x86-64 的特殊之处：
- 只有 6 个 callee-saved 寄存器（rbx, rbp, r12-r15），比 RISC-V 少一半
- TLS (FS_BASE) 不是普通寄存器，需要通过 RDMSR/WRMSR 读写
- 切换 CR3 会自动刷新 TLB（除 Global 页外）

### 3.4 LoongArch64

```rust
// arch/src/loongarch64/kcontext.rs
pub struct KContext {
    ksp: usize,              // 栈指针
    ktp: usize,              // $tp
    _sregs: [usize; 10],     // s0-s8 + s9 (10 个 callee-saved)
    kpc: usize,              // $ra
}
// 总大小: 13 * 8 = 104 bytes
```

**context_switch 汇编**：
```asm
st.d  $sp, $a0, 0*8
st.d  $tp, $a0, 1*8
st.d  $s9, $a0, 2*8      # s9 特殊处理（在 s0 前面）
st.d  $s0, $a0, 3*8
...
st.d  $s8, $a0, 11*8
st.d  $ra, $a0, 12*8

ld.d  $sp, $a1, 0*8
ld.d  $tp, $a1, 1*8
...
ld.d  $ra, $a1, 12*8
jr    $ra
```

**context_switch_pt**：
```asm
csrwr $a2, 0x19              # 写 PGDL CSR (页目录基址)
dbar  0                       # 数据屏障
invtlb 0x00, $r0, $r0        # 无效化所有 TLB
```

## 四、各架构 callee-saved 寄存器对比

| 架构 | callee-saved | 数量 | 总上下文大小 |
|------|-------------|------|-------------|
| RISC-V | s0-s11 | 12 | 120 bytes |
| AArch64 | x19-x29 | 11 | 112 bytes |
| x86-64 | rbx, rbp, r12-r15 | 6 | 72 bytes |
| LoongArch | s0-s9 | 10 | 104 bytes |

x86-64 的上下文最小（72 bytes），因为它的 ABI 只要求保存 6 个 callee-saved 寄存器，但它需要额外处理 FS_BASE MSR，增加了复杂度。

---

## 五、线程指针（TLS）机制对比

| 架构 | TLS 寄存器 | 读写方式 | 上下文切换操作 |
|------|-----------|---------|-------------|
| RISC-V | tp (x4) | 普通寄存器读写 | sd tp / ld tp |
| AArch64 | TPIDR_EL1 | 系统寄存器读写 | mrs/msr tpidr_el1 |
| x86-64 | FS_BASE | MSR 0xC0000100 | rdmsr/wrmsr |
| LoongArch | tp ($2) | 普通寄存器读写 | st.d tp / ld.d tp |

x86-64 的 TLS 最特殊——它不是寄存器而是 MSR（Model Specific Register），读写需要特权指令。

---

## 六、页表切换时的 TLB 一致性

切换页表后必须处理 TLB 一致性：

| 架构 | 页表切换指令 | TLB 处理 |
|------|-----------|---------|
| RISC-V | csrw satp | sfence.vma（必须手动刷新） |
| AArch64 | msr ttbr0_el1 | tlbi vmalle1 + dsb sy + isb |
| x86-64 | mov cr3 | 自动刷新（CR3 写入隐式刷新非 Global TLB） |
| LoongArch | csrwr PGDL | dbar 0 + invtlb 0x00 |

x86-64 是唯一不需要手动 TLB 刷新的架构（硬件自动完成），但这也意味着切换 CR3 的开销较大。

## 七、与 rcore-lab 的对比

rcore-lab 当前的上下文切换（`os/src/task/context.rs` + `os/src/task/switch.S`）：

```rust
// os/src/task/context.rs（RISC-V 硬编码）
pub struct TaskContext {
    ra: usize,
    sp: usize,
    s: [usize; 12],  // s0-s11
}
```

```asm
# os/src/task/switch.S（RISC-V 汇编）
__switch:
    sd ra, 0(a0)
    sd sp, 8(a0)
    sd s0, 16(a0)
    ...
    ld ra, 0(a1)
    ld sp, 8(a1)
    ld s0, 16(a1)
    ...
    ret
```

**改造要点**：
1. 将 `TaskContext` 重命名为 `KContext`，增加 `ktp` 字段
2. 引入 `KContextArgs` 枚举 + `Index` trait 实现
3. 将 `switch.S` 移到 `arch/src/riscv64/kcontext.rs` 中用 `#[naked]` 函数实现
4. 添加 `context_switch_pt` 变体支持页表切换
5. 添加 `read_current_tp()` 函数
